cmake_minimum_required(VERSION 3.16)
project(boustrophedon VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(BP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -no-pie")
endif()
set(BP_SOURCES src/api.c src/geometry.c src/decomposition.c src/sweep.c src/graph.c src/stitch.c)
add_library(bp_objects OBJECT ${BP_SOURCES})
set_target_properties(bp_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(bp_objects PUBLIC include PRIVATE src)
add_library(boustrophedon_static STATIC $<TARGET_OBJECTS:bp_objects>)
add_library(boustrophedon_shared SHARED $<TARGET_OBJECTS:bp_objects>)
foreach(target boustrophedon_static boustrophedon_shared)
  target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(${target} PUBLIC m)
  set_target_properties(${target} PROPERTIES OUTPUT_NAME boustrophedon)
endforeach()
set_target_properties(boustrophedon_shared PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
include(CTest)
if(BUILD_TESTING)
  foreach(suite decomposition graph sweep stitch)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${suite}.c)
      add_executable(test_${suite} tests/test_${suite}.c)
      target_include_directories(test_${suite} PRIVATE src tests)
      target_link_libraries(test_${suite} PRIVATE boustrophedon_static)
      add_test(NAME ${suite} COMMAND test_${suite})
    endif()
  endforeach()
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/examples/basic.c)
  add_executable(bp_example examples/basic.c)
  target_link_libraries(bp_example PRIVATE boustrophedon_shared)
  if(BUILD_TESTING)
    add_test(NAME public_example COMMAND bp_example)
  endif()
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/bench/runner.c)
  add_executable(bp_runner bench/runner.c)
  target_link_libraries(bp_runner PRIVATE boustrophedon_static)
  if(BUILD_TESTING)
    add_test(NAME benchmark_matrix COMMAND bp_runner --all)
    add_test(NAME integration_concave COMMAND bp_runner --size 10 --width 1 --fixture concave)
    add_test(NAME integration_obstacle COMMAND bp_runner --size 10 --width 1 --fixture obstacle --overlap 25)
    add_test(NAME invalid_benchmark_count COMMAND bp_runner --repeat -1)
    set_tests_properties(invalid_benchmark_count PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
if(BUILD_TESTING AND UNIX AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND NOT APPLE)
  add_executable(test_allocation tests/test_allocation.c)
  target_include_directories(test_allocation PRIVATE src tests)
  target_link_libraries(test_allocation PRIVATE boustrophedon_static)
  target_link_options(test_allocation PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
  add_test(NAME allocation_failures COMMAND test_allocation)
endif()
include(GNUInstallDirs)
install(TARGETS boustrophedon_static boustrophedon_shared LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/boustrophedon.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
